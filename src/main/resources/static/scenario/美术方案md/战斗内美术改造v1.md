1.2 ★充能与状态条
我方当前焦点单位（默认主角）常驻显示：
HP 条：分段式（每 10% 一格），掉血时红色残影延迟 400ms 回落，回血绿色前冲

充能格：按 needCharge 渲染 N 颗菱形宝石，满一颗亮一颗，集满时整条流光 + 技能图标弹出并可点击释放（若是手动释放）

Buff/Debuff 图标行：24px 图标 + 层数角标 + 剩余回合，对应你的 ATTR_MODIFY 效果

1.3 棋盘
透视压缩：两块棋盘各加 transform: perspective(800px) rotateX(8deg)，敌方向内倾、我方向外倾，中间交战线发光。立刻从"两张表格"变成"一个战场"。

敌我色温分离：敌方棋盘底 #2a1420→暖红雾；我方底 #141f33→冷青雾。地面纹理用同一张 tile，靠 filter: hue-rotate() 区分，省素材。

格子：虚线框改为极淡的地面网格线 + 四角短角标，空格子几乎透明，让单位成为主角。

单位占地不同 → 单位块严格按 gridW×gridH 撑满，立绘底对齐格子底边（脚踩地面），头部可超出格子上沿 20%（overflow: visible），这样大型单位才有压迫感。

单位块四件套：立绘 / 头顶细血条（2px→4px，带底槽）/ 底部椭圆投影 / 稀有度地面光环（boss 用红色脉冲环）。

交战线：中间 VS 改为一条贯穿的能量线，战斗开始时从中间向两侧扫开。

1.4 倍速与日志
1x/2x/3x/跳过 占一整行太浪费 → 收到顶栏右侧一个 » 按钮，点一下循环 1x→2x→3x，长按=跳过。省出 48px。

日志底部条改为可上滑的抽屉（56px 露 1 行 → 上滑到 70% 屏高），保留你现在弹窗里那套配色分组（我方青 / 敌方紫 / 结算金），那套做得不错，直接复用。

日志行的 └ 缩进符改为左侧色条缩进 + 层级留白，视觉更干净。

2. 战斗表现层（无素材也能先做）
按你的日志时序 释放技能 → 造成伤害 → 触发被动 → 充能+1，对应的表现节拍：
阶段	时长(1x)	表现
行动开始	200ms	行动条指针滑到该单位，棋盘上单位抬起 + 发光
前摇	250ms	攻击者向目标方向轻微后拉（蓄力）
出手	150ms	冲刺/挥击/投射物飞行
命中	100ms	目标白闪 1 帧 + 屏震（伤害越高震越狠）+ 击退位移 8px 回弹
飘字	600ms	伤害数字向上飘并淡出；暴击放大 1.4 倍 + 金色描边；治疗绿色向上
被动触发	200ms	触发者身上弹出技能名标签（如「反伤」），带一圈扩散环
充能	200ms	底部充能宝石点亮 + 音效位
死亡	500ms	单位变白 → 碎裂/溶解 → 掉落物图标弹出
性能红线：手机 6×5×2=60 格 + N 个单位，所有动画只用 transform / opacity，禁止动 width/height/top/left；特效层用单独的 .fx-layer（pointer-events:none），单位层不重排。

const { createApp } = Vue;

function queryParam(name) {
  var params = new URLSearchParams(location.search);
  return params.get(name) || '';
}

createApp({
  data() {
    return {
      loading: true,
      error: '',
      levelId: '',
      levelTitle: '',
      stageTree: [],
      monsters: [],
      allies: [],
      battleStarted: false,
      fighting: false,
      replaying: false,
      outcome: '',
      visibleLogs: [],
      pendingLogs: [],
      actingName: '',
      hitName: '',
      deadNames: {},
      hpMap: {},
      replayTimer: null,
      replayData: null,
      replayLogs: [],
      replayIndex: 0,
      skipRequested: false,
      playbackSpeed: 1,
      resultModal: { show: false, outcome: '', ticks: null, drops: [] },
      logModal: false,
      toast: { show: false, text: '', type: 'error' }
    };
  },
  computed: {
    showPlaybackBar() {
      return this.battleStarted && (this.fighting || this.replaying) && !this.resultModal.show;
    },
    canFight() {
      return this.monsters.length > 0 && this.allies.length > 0 && !this.fighting && !this.replaying;
    },
    statusLabel() {
      if (this.outcome === 'WIN') return '胜利';
      if (this.outcome === 'LOSE') return '失败';
      if (this.fighting || this.replaying) return '战斗中';
      return '待命';
    },
    statusPillClass() {
      if (this.outcome === 'WIN') return 'win';
      if (this.outcome === 'LOSE') return 'lose';
      if (this.fighting || this.replaying) return 'fighting';
      return 'ready';
    },
    battleTip() {
      if (!this.monsters.length) return '该关卡暂无怪物配置';
      if (!this.allies.length) return '请先在「布局」上阵角色';
      return '双方已就位，可开始行动值战斗';
    },
    nextLevelId() {
      var list = this.flattenLevels(this.stageTree);
      for (var i = 0; i < list.length; i++) {
        if (list[i].id === this.levelId) {
          return list[i + 1] ? list[i + 1].id : '';
        }
      }
      return '';
    },
    allyNameSet() {
      var set = {};
      (this.allies || []).forEach((a) => {
        var n = a && (a.roleName || a.name);
        if (n) set[n] = true;
      });
      // 开战快照里的己方名也纳入，保证日志着色
      Object.keys(this.hpMap || {}).forEach((k) => {
        var s = this.hpMap[k];
        if (s && (s.side === 'ALLY' || s.side === 'ally')) set[k] = true;
      });
      return set;
    },
    enemyNameSet() {
      var set = {};
      (this.monsters || []).forEach((m) => {
        var n = m && (m.monsterName || m.name);
        if (n) set[n] = true;
      });
      Object.keys(this.hpMap || {}).forEach((k) => {
        var s = this.hpMap[k];
        if (s && (s.side === 'ENEMY' || s.side === 'enemy')) set[k] = true;
      });
      return set;
    },
    /** 一组行动包成一块；尾部胜负+掉落用固定色整块括起 */
    visibleLogBlocks() {
      var blocks = [];
      var group = -1;
      var current = null;
      var lines = this.visibleLogs || [];
      for (var i = 0; i < lines.length; i++) {
        var line = lines[i] || '';
        var child = line.indexOf('└') >= 0 || /^\s+/.test(line);
        var isBoot = line.indexOf('装备充能技能') >= 0;
        var isResultHead = line.indexOf('战斗胜利') >= 0 || line.indexOf('战斗失败') >= 0
          || line.indexOf('总计掉落') >= 0;
        if (isBoot) {
          current = null;
          blocks.push({
            kind: 'special',
            cls: this.logLineClass(line),
            html: this.escapeHtml(line)
          });
          continue;
        }
        if (isResultHead) {
          current = {
            kind: 'group',
            groupCls: 'log-group-result',
            rows: []
          };
          blocks.push(current);
          current.rows.push({
            cls: [this.logLineClass(line), 'is-head'].join(' '),
            html: this.escapeHtml(line)
          });
          continue;
        }
        if (!child) {
          group += 1;
          var g = group % 4;
          current = {
            kind: 'group',
            groupCls: 'log-group-' + g,
            rows: []
          };
          blocks.push(current);
          current.rows.push({
            cls: [this.logLineClass(line), 'is-head'].join(' '),
            html: this.formatLogHtml(line)
          });
        } else {
          if (!current) {
            group += 1;
            current = {
              kind: 'group',
              groupCls: 'log-group-' + (group % 4),
              rows: []
            };

    formatLogHtml(line) {
      var raw = String(line == null ? '' : line);

      var body = raw;
      // 新格式：首行「行动值 N」换行 + 正文；兼容旧「[行动值 N] 正文」
      var avMulti = raw.match(/^行动值\s*(\d+)\n([\s\S]*)$/);
      var avLegacy = raw.match(/^\[行动值\s*(\d+)\]\s*(.*)$/);
      if (avMulti) {
        prefix = '<div class="log-action-val">行动值 ' + this.escapeHtml(avMulti[1]) + '</div>';
        body = avMulti[2] || '';
      } else if (avLegacy) {
        prefix = '<div class="log-action-val">行动值 ' + this.escapeHtml(avLegacy[1]) + '</div>';
        body = avLegacy[2] || '';
      }
      // 对 {目标} 造成 {N} 伤害 → 伤害数字按阵营上色；己方目标显示为 [名]
      var m = body.match(/^(.*对\s*)(.+?)(\s*造成\s*)(\d+)(\s*伤害.*)$/);
      if (m) {
        var target = m[2];
        var colorCls = 'dmg-num';
        if (this.allyNameSet[target]) colorCls = 'dmg-num-ally';
        else if (this.enemyNameSet[target]) colorCls = 'dmg-num-enemy';
        return prefix
          + '<div class="log-action-body">'
          + this.wrapNamesInSegment(m[1])
          + this.formatLogNameHtml(target)
          + this.escapeHtml(m[3])
          + '<span class="' + colorCls + '">' + this.escapeHtml(m[4]) + '</span>'
          + this.wrapNamesInSegment(m[5])
          + '</div>';
      }
      // 释放行等：己方 [角色名] 着色；怪物名加粗
      if (prefix) {
        return prefix + '<div class="log-action-body">' + this.wrapNamesInSegment(body) + '</div>';
      }
      return this.wrapNamesInSegment(body);

        if (!name || raw.indexOf(name) < 0) return;
        var re = new RegExp(this.escapeRegExp(this.escapeHtml(name)), 'g');

      });
      return out;

    wrapEnemyNamesInHtml(html, raw) {
      // 先长后短，避免 史莱姆 吃掉 史莱姆1；跳过已包裹片段
      var names = Object.keys(this.enemyNameSet || {}).sort(function (a, b) { return b.length - a.length; });
      var out = html;
      names.forEach((name) => {
        if (!name || raw.indexOf(name) < 0) return;
        var token = this.escapeHtml(name);
        var re = new RegExp('(?!<[^>]*)' + this.escapeRegExp(token) + '(?![^<]*>)', 'g');

          // 已在 log-name-* 内则不重复包
          var before = full.slice(0, offset);
          if (/<span class="log-name-(?:ally|enemy)">[^<]*$/.test(before)) return match;
          return '<span class="log-name-enemy">[' + match + ']</span>';
        });
      });
      return out;
    },

      if (!this.replaying) return;
      if (this.replayIndex >= this.replayLogs.length) {
        this.pendingLogs = [];
        this.finishReplay(this.replayData || {});
        return;
      }
      var line = this.replayLogs[this.replayIndex++];
      this.pendingLogs = this.replayLogs.slice(this.replayIndex);
      this.visibleLogs.push(line);
      this.applyLogAnim(line);
      this.$nextTick(() => {
        var el = this.$refs.logList;
        if (el) el.scrollTop = el.scrollHeight;
      });
      this.scheduleReplayStep();
    },
    skipReplay() {
      if (!this.fighting && !this.replaying) return;
      if (this.fighting && !this.replaying) {
        this.skipRequested = true;
        return;
      }
      if (this.replayTimer) {
        clearTimeout(this.replayTimer);
        this.replayTimer = null;
      }
      var rest = this.replayLogs.slice(this.replayIndex);
      if (rest.length) {
        this.visibleLogs = this.visibleLogs.concat(rest);
        var dead = {};
        Object.keys(this.deadNames).forEach((k) => { dead[k] = true; });
        rest.forEach((line) => {
          this.applyHpFromLog(line);
          var dmg = line && line.match(/对\s*(.+?)\s*造成/);
          if (dmg && line.indexOf('击杀') >= 0) {
            dead[dmg[1].trim()] = true;
          }
        });
        this.deadNames = dead;
      }
      this.replayIndex = this.replayLogs.length;
      this.pendingLogs = [];
      this.actingName = '';
      this.hitName = '';
      this.finishReplay(this.replayData || {});
    },
    applyLogAnim(line) {
      if (!line) return;
      this.applyHpFromLog(line);
      var body = this.stripLogActionPrefix(line);
      var cast = body.match(/^(.+?)\s*释放/);
      if (cast) {
        this.actingName = cast[1].trim();
        this.hitName = '';
        var self = this;
        var name = cast[1].trim();
        setTimeout(function () { if (self.actingName === name) self.actingName = ''; }, this.animDuration(420));
        return;
      }
      var dmg = body.match(/对\s*(.+?)\s*造成/);
      if (dmg) {
        this.hitName = dmg[1].trim();
        if (body.indexOf('击杀') >= 0) {
          var dead = {};
          Object.keys(this.deadNames).forEach((k) => { dead[k] = true; });
          dead[this.hitName] = true;
          this.deadNames = dead;
        }
        var self2 = this;
        var hit = dmg[1].trim();
        setTimeout(function () { if (self2.hitName === hit) self2.hitName = ''; }, this.animDuration(380));
      }
    },
    finishReplay(data) {
      if (this.replayTimer) {
        clearTimeout(this.replayTimer);
        this.replayTimer = null;
      }
      this.replaying = false;
      this.fighting = false;
      this.skipRequested = false;
      this.actingName = '';
      this.hitName = '';
      this.outcome = (data && data.outcome) || '';
      this.resultModal = {

        outcome: this.outcome,

      };
    },
    closeResultModal() {

    },
    openLogModal() {

      });
    },

    },

    },
